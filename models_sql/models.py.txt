from datetime import datetime
from flask_sqlalchemy import SQLAlchemy

db = SQLAlchemy()

# Note: class db viết hoa chữ đầu, số ít
# table name số nhiều
parent_kid = db.Table(
	"parent_kid",
	db.Column("parent_id", db.Integer, db.ForeignKey("parents.id"), primary_key=True),
 	db.Column("kid_id", db.Integer, db.ForeignKey("kids.id"), primary_key=True)
)

class Parent(db.Model):
    __tablename__ = "parents"
    id = db.Column(db.Integer, primary_key=True, autoincrement=True)
    original_doc_id = db.Column(db.Integer, nullable = True)
    name = db.Column(db.String(50))
    phone = db.Column(db.String(15))
    address = db.Column(db.String(100))
    note = db.Column(db.String(50))
    is_deleted = db.Column(db.Boolean, default=False)

    kids = db.relationship(
        "Kid", back_populates="parents", secondary=parent_kid)
    exams = db.relationship("Exam", back_populates="parent", cascade="all, delete-orphan")
    
class Kid(db.Model):
    __tablename__ = 'kids'
    id = db.Column(db.Integer, primary_key=True, autoincrement=True)
    name = db.Column(db.String(50))
    birthday = db.Column(db.String(15))
    note = db.Column(db.String(30))
    is_deleted = db.Column(db.Boolean, default=False)

    parents =  db.relationship("Parent", back_populates="kids", secondary=parent_kid)
    exams = db.relationship("Exam", back_populates="kid", cascade="all, delete-orphan")

class Drug(db.Model):
    __tablename__ = "drugs"
    id = db.Column(db.Integer, primary_key=True, autoincrement=True)
    sku = db.Column(db.String(50), nullable = True)
    name = db.Column(db.String(50))
    cost_per_unit = db.Column(db.Numeric)
    sell_price_per_unit = db.Column(db.Numeric)
    stock = db.Column(db.Integer, nullable=True)
    is_deleted = db.Column(db.Boolean, default=False)

    exam_usages = db.relationship("ExamDrug", back_populates="drug", cascade="all, delete-orphan")
    purchase_items = db.relationship("PurchaseItem", back_populates="drug", cascade="all, delete-orphan")

    
class Exam(db.Model):
    __tablename__ = "exams"
    id = db.Column(db.String(64), primary_key=True)
    parent_id = db.Column(db.Integer, db.ForeignKey("parents.id"), nullable=False)
    kid_id = db.Column(db.Integer, db.ForeignKey("kids.id"), nullable=True)
    department = db.Column(db.String)
    exam_date = db.Column(db.String(15))
    create_at = db.Column(db.DateTime, default=datetime.utcnow)
    edit_at = db.Column(db.DateTime, default=datetime.utcnow)
    kid_weight = db.Column(db.String(10))
    kid_height = db.Column(db.String(10))
    exam_note = db.Column(db.Text)
    service_type = db.Column(db.String(50))
    service_fee = db.Column(db.Numeric, nullable=True)
    next_exam_date = db.Column(db.String(30), nullable=True)
    paid_status = db.Column(db.Boolean, default=False)
    paid_date = db.Column(db.DateTime, default=datetime.utcnow, nullable=True)
    is_deleted = db.Column(db.Boolean, default=False)

    exam_drugs = db.relationship("ExamDrug", back_populates="exam", cascade="all, delete-orphan")
    parent = db.relationship("Parent", back_populates="exams")
    kid = db.relationship("Kid", back_populates="exams")


class ExamDrug(db.Model):
    __tablename__ = "exam_drugs"
    id = db.Column(db.Integer, primary_key=True, autoincrement=True)
    exam_id = db.Column(db.String(64), db.ForeignKey("exams.id", ondelete="CASCADE"), nullable=False, index=True)
    drug_id = db.Column(db.Integer, db.ForeignKey("drugs.id"), nullable=True)
    name = db.Column(db.String(50), nullable=True)
    quantity = db.Column(db.Integer, nullable = True)
    sell_price_per_unit = db.Column(db.Numeric, nullable=True)
    subtotal = db.Column(db.Numeric)
    note = db.Column(db.String(100), nullable=True)
    create_at = db.Column(db.DateTime, default=datetime.utcnow)
    update_at = db.Column(db.DateTime, default=datetime.utcnow)
    
    exam = db.relationship("Exam", back_populates="exam_drugs")
    drug = db.relationship("Drug", back_populates="exam_usages")


class Purchase(db.Model):
    __tablename__ = "purchases"
    id = db.Column(db.Integer, primary_key=True, autoincrement=True)
    order_date = db.Column(db.DateTime, default=datetime.utcnow)
    paid_status = db.Column(db.Boolean, default=False)
    paid_date = db.Column(db.DateTime, default=datetime.utcnow, nullable=True)
    note = db.Column(db.Text)
    total_cost = db.Column(db.Numeric)
    create_at = db.Column(db.DateTime, default=datetime.utcnow)
    update_at = db.Column(db.DateTime, nullable=True)
    is_deleted = db.Column(db.Boolean, default=False)

    items = db.relationship("PurchaseItem", back_populates="purchase", cascade = "all, delete-orphan")    

class PurchaseItem(db.Model):
    __tablename__ = "purchase_items"
    id = db.Column(db.Integer, primary_key=True, autoincrement=True)
    purchase_id = db.Column(db.Integer, db.ForeignKey("purchases.id"), nullable=False)
    name = db.Column(db.String(256))
    quantity = db.Column(db.Integer)
    cost_subtotal = db.Column(db.Numeric)
    note = db.Column(db.Text, nullable=True)
    cost_per_unit = db.Column(db.Numeric, nullable=True)

    purchase = db.relationship("Purchase", back_populates="items")
    drug = db.relationship("Drug", back_populates="purchase_items")
